101constexpr uint16_t LdoCodeToMillivolts(uint8_t code, bool is_ldo2) noexcept {
102 const uint8_t c = static_cast<uint8_t>(code & 0x1FU);
103 if (is_ldo2) {
104 if (c <= 0x0FU) {
105 return static_cast<uint16_t>(1800U + 100U * c);
106 }
107 return 0U;
108 }
109 if (c <= 0x0FU) {
110 return static_cast<uint16_t>(750U + 50U * c);
111 }
112 return static_cast<uint16_t>(1800U + 100U * (c - 0x10U));

141constexpr uint16_t VbusLimitRegToMilliamps(uint8_t reg_value) noexcept {
142 const uint8_t code = static_cast<uint8_t>((reg_value >> 3U) & 0x1FU);
143 // First 9 entries: 10, 15, 20, 25, 30, 35, 40, 45, 50 mA (5 mA step from 10).
144 if (code <= 0x08U) {
145 return static_cast<uint16_t>(10U + 5U * code);
146 }
147 // 0x09..0x0F: 100, 150, 200, 300, 400, 500, 600 mA.
148 switch (code) {
149 case 0x09: return 100;
150 case 0x0A: return 150;
151 case 0x0B: return 200;
152 case 0x0C: return 300;
153 case 0x0D: return 400;
154 case 0x0E: return 500;
155 case 0x0F: return 600;
156 case 0x10: return 700;
157 case 0x11: return 800;
158 case 0x12: return 900;
159 case 0x13: return 1000;
160 case 0x14: return 1500; // Portenta default
161 case 0x15: return 1600;
162 case 0x16: return 1700;
163 case 0x17: return 1800;
164 default: return 0; // codes above table reserved
165 }
166}

174constexpr uint8_t VbusLimitMilliampsToReg(uint16_t mA) noexcept {
175 uint8_t code = 0xFFU;
176 switch (mA) {
177 case 10: code = 0x00; break;
178 case 100: code = 0x09; break;
179 case 500: code = 0x0E; break;
180 case 1000: code = 0x13; break;
181 case 1500: code = 0x14; break; // Portenta default
182 }
183 return (code == 0xFFU) ? 0xFFU : static_cast<uint8_t>(code << 3U);
184}
