273template <typename BusType>
274bool PF1550<BusType>::ReadLatchedFaults(FaultFlags& faults) noexcept {
275 faults = FaultFlags{};
276 uint8_t sw0 = 0, sw1 = 0, ldo0 = 0, temp0 = 0, misc0 = 0, chg = 0, vbus = 0;
277 bool any_fail = false;
278
279 any_fail |= !readReg8(static_cast<uint8_t>(Register::SwIntStat0), sw0);
280 any_fail |= !readReg8(static_cast<uint8_t>(Register::SwIntStat1), sw1);
281 any_fail |= !readReg8(static_cast<uint8_t>(Register::LdoIntStat0), ldo0);
282 any_fail |= !readReg8(static_cast<uint8_t>(Register::TempIntStat0), temp0);
283 any_fail |= !readReg8(static_cast<uint8_t>(Register::MiscIntStat0), misc0);
284 any_fail |= !readReg8(static_cast<uint8_t>(Register::ChgInt), chg);
285 any_fail |= !readReg8(static_cast<uint8_t>(Register::VbusSns), vbus);
286
287 if (sw0 & SwIntStat0Bits::kSw1Ls) faults.Set(FaultFlags::kSw1Ls);
288 if (sw0 & SwIntStat0Bits::kSw2Ls) faults.Set(FaultFlags::kSw2Ls);
289 if (sw0 & SwIntStat0Bits::kSw3Ls) faults.Set(FaultFlags::kSw3Ls);
290 if (sw1 & SwIntStat1Bits::kSw1Hs) faults.Set(FaultFlags::kSw1Hs);
291 if (sw1 & SwIntStat1Bits::kSw2Hs) faults.Set(FaultFlags::kSw2Hs);
292 if (sw1 & SwIntStat1Bits::kSw3Hs) faults.Set(FaultFlags::kSw3Hs);
293 if (ldo0 & LdoIntStat0Bits::kLdo1Fault) faults.Set(FaultFlags::kLdo1Fault);
294 if (ldo0 & LdoIntStat0Bits::kLdo2Fault) faults.Set(FaultFlags::kLdo2Fault);
295 if (ldo0 & LdoIntStat0Bits::kLdo3Fault) faults.Set(FaultFlags::kLdo3Fault);
296 if (temp0 & TempIntStat0Bits::kTempWarn) faults.Set(FaultFlags::kTempWarn);
297 if (temp0 & TempIntStat0Bits::kTempShdn) faults.Set(FaultFlags::kTempShdn);
298 if (misc0 & MiscIntStat0Bits::kPwrOn) faults.Set(FaultFlags::kPwrOn);
299 if (misc0 & MiscIntStat0Bits::kVsysLow) faults.Set(FaultFlags::kVsysLow);
300 if (misc0 & MiscIntStat0Bits::kVsysOv) faults.Set(FaultFlags::kVsysOv);
301 if (misc0 & MiscIntStat0Bits::kWdiAssert) faults.Set(FaultFlags::kWdiAssert);
302 if (misc0 & MiscIntStat0Bits::kResetBmcu) faults.Set(FaultFlags::kResetBmcu);
303 if (chg != 0) faults.Set(FaultFlags::kChgFault);
304 // VBUS_VALID is bit 5 of VBUS_SNS — invalid means !bit5.
305 if ((vbus & (1u << 5)) == 0) faults.Set(FaultFlags::kVbusInval);
306
307 if (any_fail) {
308 faults.Set(FaultFlags::kI2cReadFail);
309 setError(Error::DiagnosticRead);
310 return false;
311 }
312 return true;
313}

349template <typename BusType>
350bool PF1550<BusType>::fillRailSnapshot(DiagnosticSnapshot& snap) noexcept {
351 struct RailMap {
352 RailId id;
353 Register volt_reg;
354 Register ctrl_reg;
355 bool is_ldo2; // true → LDO2 group B encoding
356 bool is_sw;
357 };
358 static constexpr RailMap kMap[] = {
359 {RailId::Sw1, Register::Sw1Volt, Register::Sw1Ctrl, false, true},
360 {RailId::Sw2, Register::Sw2Volt, Register::Sw2Ctrl, false, true},
361 {RailId::Sw3, Register::Sw3Volt, Register::Sw3Ctrl, false, true},
362 {RailId::Ldo1, Register::Ldo1Volt, Register::Ldo1Ctrl, false, false},
363 {RailId::Ldo2, Register::Ldo2Volt, Register::Ldo2Ctrl, true, false},
364 {RailId::Ldo3, Register::Ldo3Volt, Register::Ldo3Ctrl, false, false},
365 };
366 bool ok = true;
367 for (const auto& m : kMap) {
368 auto& d = snap.rails[static_cast<size_t>(m.id)];
369 if (!readReg8(static_cast<uint8_t>(m.volt_reg), d.volt_reg)) { ok = false; continue; }
370 if (!readReg8(static_cast<uint8_t>(m.ctrl_reg), d.ctrl_reg)) { ok = false; continue; }
371 if (m.id == RailId::Sw3) {
372 d.set_voltage_mv = Sw3CodeToMillivolts(d.volt_reg); /* Table 37 */
373 } else if (m.is_sw) {
374 const bool dvs = sw_dvs_enabled_[m.id == RailId::Sw1 ? 0U : 1U];
375 d.set_voltage_mv = dvs ? SwDvsCodeToMillivolts(d.volt_reg) : SwCodeToMillivolts(d.volt_reg);
376 } else {
377 d.set_voltage_mv = LdoCodeToMillivolts(d.volt_reg & 0x1FU, m.is_ldo2);
378 }
379 d.enabled_run = (d.ctrl_reg & SwCtrlBits::kEn) != 0;
380 d.enabled_stby = (d.ctrl_reg & SwCtrlBits::kStbyEn) != 0;
381 }
382 return ok;
383}

385template <typename BusType>
386bool PF1550<BusType>::fillFaultSnapshot(DiagnosticSnapshot& snap) noexcept {
387 if (!ReadLatchedFaults(snap.faults)) {
388 return false;
389 }
390 // Per-rail latched + live mapping.
391 uint8_t sw_sense0 = 0, sw_sense1 = 0, ldo_sense = 0;
392 (void)readReg8(static_cast<uint8_t>(Register::SwIntSense0), sw_sense0);
393 (void)readReg8(static_cast<uint8_t>(Register::SwIntSense1), sw_sense1);
394 (void)readReg8(static_cast<uint8_t>(Register::LdoIntSense0), ldo_sense);
395
396 auto mark = [&](RailId id, bool latched, bool live) {
397 auto& d = snap.rails[static_cast<size_t>(id)];
398 d.stat_latched = latched;
399 d.sense_live = live;
400 };
401 mark(RailId::Sw1,
402 snap.faults.Has(FaultFlags::kSw1Ls | FaultFlags::kSw1Hs),
403 ((sw_sense0 & SwIntStat0Bits::kSw1Ls) || (sw_sense1 & SwIntStat1Bits::kSw1Hs)) != 0);
404 mark(RailId::Sw2,
405 snap.faults.Has(FaultFlags::kSw2Ls | FaultFlags::kSw2Hs),
406 ((sw_sense0 & SwIntStat0Bits::kSw2Ls) || (sw_sense1 & SwIntStat1Bits::kSw2Hs)) != 0);
407 mark(RailId::Sw3,
408 snap.faults.Has(FaultFlags::kSw3Ls | FaultFlags::kSw3Hs),
409 ((sw_sense0 & SwIntStat0Bits::kSw3Ls) || (sw_sense1 & SwIntStat1Bits::kSw3Hs)) != 0);
410 mark(RailId::Ldo1, snap.faults.Has(FaultFlags::kLdo1Fault),
411 (ldo_sense & LdoIntStat0Bits::kLdo1Fault) != 0);
412 mark(RailId::Ldo2, snap.faults.Has(FaultFlags::kLdo2Fault),
413 (ldo_sense & LdoIntStat0Bits::kLdo2Fault) != 0);
414 mark(RailId::Ldo3, snap.faults.Has(FaultFlags::kLdo3Fault),
415 (ldo_sense & LdoIntStat0Bits::kLdo3Fault) != 0);
416 return true;
417}

419template <typename BusType>
420bool PF1550<BusType>::ReadDiagnosticSnapshot(DiagnosticSnapshot& out) noexcept {
421 out = DiagnosticSnapshot{};
422 bool ok = true;
423 ok &= readReg8(static_cast<uint8_t>(Register::DeviceId), out.device_id);
424 ok &= readReg8(static_cast<uint8_t>(Register::OtpFlavor), out.otp_flavor);
425 ok &= readReg8(static_cast<uint8_t>(Register::SiliconRev), out.silicon_rev);
426 ok &= readReg8(static_cast<uint8_t>(Register::IntCategory), out.int_category);
427 ok &= readReg8(static_cast<uint8_t>(Register::StateInfo), out.state_info_reg);
428 out.state = DecodeStateInfo(out.state_info_reg);
429
430 uint8_t chg_raw = 0;
431 if (readReg8(static_cast<uint8_t>(Register::ChgSense), chg_raw)) {
432 out.chg_sense_reg = chg_raw;
433 out.charger = DecodeChargerSense(chg_raw);
434 } else {
435 ok = false;
436 }
437 ok &= readReg8(static_cast<uint8_t>(Register::VbusSns), out.vbus_sns_reg);
438
439 uint8_t vbus_lim_reg = 0;
440 if (readReg8(static_cast<uint8_t>(Register::VbusInCurrentLimit), vbus_lim_reg)) {
441 out.vbus_in_limit_ma = VbusLimitRegToMilliamps(vbus_lim_reg);
442 }
443
444 ok &= fillRailSnapshot(out);
445 ok &= fillFaultSnapshot(out);
446
447 out.read_ok = ok;
448 if (!ok) {
449 setError(Error::DiagnosticRead);
450 }
451 return ok;
452}

476template <typename BusType>
477bool PF1550<BusType>::RunPowerSelfTest(SelfTestResult& out) noexcept {
478 out = SelfTestResult{};
479 out.ran = true;
480
481 if (!ReadDiagnosticSnapshot(out.snapshot)) {
482 out.worst_severity = FaultSeverity::kMcuKill; // could not even read PMIC
483 setError(Error::SelfTestFailed);
484 return false;
485 }
486
487 out.device_id_ok = (out.snapshot.device_id == kExpectedDeviceId);
488 out.state_run = (out.snapshot.state == PmicState::Run);
489
490 out.all_rails_enabled = true;
491 for (size_t i = 0; i < static_cast<size_t>(RailId::COUNT); ++i) {
492 // Skip VSNVS (no `EN` bit in same place).
493 if (static_cast<RailId>(i) == RailId::Vsnvs) continue;
494 if (!out.snapshot.rails[i].enabled_run) {
495 out.all_rails_enabled = false;
496 break;
497 }
498 }
499
500 out.faults = out.snapshot.faults;
501 out.worst_severity = WorstSeverityPortentaH7(out.faults);
502
503 auto sev_val = [](FaultSeverity s) noexcept {
504 return static_cast<uint8_t>(s);
505 };
506 auto raise = [&](FaultSeverity target) noexcept {
507 if (sev_val(out.worst_severity) < sev_val(target)) {
508 out.worst_severity = target;
509 }
510 };
511
512 if (!out.device_id_ok) raise(FaultSeverity::kMcuKill);
513 if (!out.state_run) raise(FaultSeverity::kCritical);
514 if (!out.all_rails_enabled) raise(FaultSeverity::kCritical);
515
516 if (out.worst_severity != FaultSeverity::kInfo) {
517 setError(Error::SelfTestFailed);
518 }
519 return out.worst_severity == FaultSeverity::kInfo;
520}
