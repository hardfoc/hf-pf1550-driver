176constexpr RailId FaultRail(uint32_t single_flag) noexcept {
177 if (single_flag == FaultFlags::kSw1Ls || single_flag == FaultFlags::kSw1Hs) return RailId::Sw1;
178 if (single_flag == FaultFlags::kSw2Ls || single_flag == FaultFlags::kSw2Hs) return RailId::Sw2;

193constexpr FaultSeverity FaultSeverityPortentaH7(uint32_t single_flag) noexcept {
194 switch (single_flag) {
195 // MCU is powered from SW3 → losing SW3 kills the MCU.
196 case FaultFlags::kSw3Hs:
197 case FaultFlags::kSw3Ls:
198 // LDO1 (1.0 V) feeds the SOC core analog and MIPI bridge — losing it
199 // crashes the platform.
200 case FaultFlags::kLdo1Fault:
201 // LDO3 (1.2 V) feeds Ethernet PHY VDDCR and DSI core.
202 case FaultFlags::kLdo3Fault:
203 case FaultFlags::kVsysLow:
204 return FaultSeverity::kMcuKill;
205
206 // SW1 (+3V1SW) feeds USB/ULPI, SDRAM, ETH PHY and LDO inputs — losing it
207 // takes USB / network down but MCU can still run on SW3+SW3-derived
208 // domains. It IS however necessary for LDO outputs.
209 case FaultFlags::kSw1Hs:
210 case FaultFlags::kSw1Ls:
211 // SW2 feeds carrier +3V3 only; MCU keeps running.
212 case FaultFlags::kSw2Hs:
213 case FaultFlags::kSw2Ls:
214 // LDO2 (1.8 V) for USB PHY — USB dies, MCU survives.
215 case FaultFlags::kLdo2Fault:
216 case FaultFlags::kTempShdn:
217 case FaultFlags::kBattOv:
218 case FaultFlags::kVsysOv:
219 return FaultSeverity::kCritical;
220
221 case FaultFlags::kTempWarn:
222 case FaultFlags::kChgFault:
223 case FaultFlags::kVbusInval:
224 case FaultFlags::kWdiAssert:
225 case FaultFlags::kI2cReadFail:
226 case FaultFlags::kI2cWriteFail:
227 return FaultSeverity::kWarning;
228
229 case FaultFlags::kPwrOn:
230 case FaultFlags::kResetBmcu:
231 default:
232 return FaultSeverity::kInfo;
233 }
234}

239constexpr FaultSeverity WorstSeverityPortentaH7(FaultFlags fl) noexcept {
240 FaultSeverity worst = FaultSeverity::kInfo;
241 for (uint32_t bit = 0; bit < 32; ++bit) {
242 const uint32_t mask = 1u << bit;
243 if ((fl.value & mask) == 0) continue;
244 const auto sev = FaultSeverityPortentaH7(mask);
245 if (static_cast<uint8_t>(sev) > static_cast<uint8_t>(worst)) {
246 worst = sev;
247 }
248 }
249 return worst;
250}
