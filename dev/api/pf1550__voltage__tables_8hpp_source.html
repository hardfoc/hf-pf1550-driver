118constexpr uint16_t LdoCodeToMillivolts(uint8_t code, bool is_ldo2) noexcept {
119 const uint8_t c = static_cast<uint8_t>(code & 0x1FU);
120 if (is_ldo2) {
121 if (c <= 0x0FU) {

158constexpr uint16_t VbusLimitRegToMilliamps(uint8_t reg_value) noexcept {
159 const uint8_t code = static_cast<uint8_t>((reg_value >> 3U) & 0x1FU);
160 // First 9 entries: 10, 15, 20, 25, 30, 35, 40, 45, 50 mA (5 mA step from 10).
161 if (code <= 0x08U) {
162 return static_cast<uint16_t>(10U + 5U * code);
163 }
164 // 0x09..0x0F: 100, 150, 200, 300, 400, 500, 600 mA.
165 switch (code) {
166 case 0x09: return 100;
167 case 0x0A: return 150;
168 case 0x0B: return 200;
169 case 0x0C: return 300;
170 case 0x0D: return 400;
171 case 0x0E: return 500;
172 case 0x0F: return 600;
173 case 0x10: return 700;
174 case 0x11: return 800;
175 case 0x12: return 900;
176 case 0x13: return 1000;
177 case 0x14: return 1500; // Portenta default
178 default: return 0; // codes 21..31 are reserved (Table 58)
179 }
180}

188constexpr uint8_t VbusLimitMilliampsToReg(uint16_t mA) noexcept {
189 uint8_t code = 0xFFU;
190 switch (mA) {
191 case 10: code = 0x00; break;
192 case 100: code = 0x09; break;
193 case 500: code = 0x0E; break;
194 case 1000: code = 0x13; break;
195 case 1500: code = 0x14; break; // Portenta default
196 }
197 return (code == 0xFFU) ? 0xFFU : static_cast<uint8_t>(code << 3U);
198}
